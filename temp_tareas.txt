function initCardInteractions() {
  document.querySelectorAll('.task-card-toggle').forEach(toggle => {
    toggle.addEventListener('click', (e) => {
      e.stopPropagation();
      const id = toggle.dataset.toggle;
      const items = document.getElementById(`items-${id}`);
      const isExp = items.classList.contains('expanded');
      if (isExp) {
        items.classList.remove('expanded');
        toggle.classList.remove('expanded');
      } else {
        items.classList.add('expanded');
        toggle.classList.add('expanded');
      }
    });
  });

  document.querySelectorAll('.task-item-checkbox').forEach(box => {
    box.addEventListener('click', (e) => {
      e.stopPropagation();
      box.classList.toggle('checked');
      const id = box.dataset.check;
      const text = document.querySelector(`.task-item-text[data-text="${id}"]`);
      if (box.classList.contains('checked')) {
        text.classList.add('checked');
        box.textContent = 'âœ“';
      } else {
        text.classList.remove('checked');
        box.textContent = '';
      }
    });
  });

  document.querySelectorAll('.task-add-action').forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.stopPropagation();
      alert('Agregar acciÃ³n (funcionalidad prÃ³ximamente)');
    });
  });
}
